//Question_16. Write a query to list departments that have no employees. & create a view for those salespeople who belong to the city of Mumbai. 
SELECT d.Department_ID, d.department_Name
FROM Department d
LEFT JOIN Employee e
ON d.Department_ID = e.Department_ID
WHERE e.Employee_Id IS NULL;

create view Mumbai_Employees As
select e.Employee_ID,e.Employee_Name,e.Employee_Salary,d.Department_Name,d.Employee_Location
from Employee e
join Department d
on e.Department_ID=d.Department_ID
where d.Employee_Location='Mumbai';
